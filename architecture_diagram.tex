\begin{tikzpicture}[

    clientBox/.style={card, fill=clientBg, draw=clientBorder, text=clientText, minimum width=4.8cm},
    connArrow/.style={->, line width=1.3pt, draw=gray!65},
    dataArrow/.style={->, line width=1.3pt, draw=jobBorder, dashed},
    httpArrow/.style={<->, line width=1.4pt, draw=serveBorder}

    % Top Row Nodes
    \node[configBox] (config) {
        \textbf{\large ConfigMap}\\
        \texttt{training-config}\\[2pt]

    };

    \node[pvcBox, right=4.2cm of config] (pvc) {

    };

    \node[serveBox, below=2.4cm of pvc] (serve) {

    % Bottom Serving Stack Nodes
    \node[svcBox, below=1.8cm of serve] (svc) {

        \texttt{model-serving}\\[2pt]
        \footnotesize Port 80 $\rightarrow$ TargetPort 8080
    };

        \textbf{\large Client / User App}\\
